2. **tests.cpp** - Test source (~60-100 lines):
```cpp
#define CATCH_CONFIG_MAIN
#include <catch2/catch.hpp>

// Basic math tests
TEST_CASE("Math operations", "[math]") {
    SECTION("addition") {
        REQUIRE(1 + 1 == 2);
        REQUIRE(2 + 2 == 4);
    }

    SECTION("multiplication") {
        REQUIRE(2 * 3 == 6);
        REQUIRE(4 * 5 == 20);
    }
}

// String tests
TEST_CASE("String operations", "[string]") {
    std::string hello = "Hello";
    std::string world = "World";

    REQUIRE(hello.length() == 5);
    REQUIRE(hello + " " + world == "Hello World");
}

// C++20 concepts test (demonstrates modern C++)
#include <concepts>

template<typename T>
concept Numeric = std::integral<T> || std::floating_point<T>;

template<Numeric T>
T add(T a, T b) { return a + b; }

TEST_CASE("C++20 concepts", "[cpp20]") {
    REQUIRE(add(1, 2) == 3);
    REQUIRE(add(1.5, 2.5) == 4.0);
}
```

The tests.cpp must:
- Define CATCH_CONFIG_MAIN before including catch.hpp
- Include multiple TEST_CASE blocks with SECTION
- Use REQUIRE macros for assertions
- Include at least one C++20 feature (concepts)
- All tests must pass (no intentional failures)
