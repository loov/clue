3. **utils.cpp** - Implementation using simplemath:
```cpp
#include "utils.h"
#include "simplemath/math.h"
#include <sstream>

namespace stringutils {
    std::string format_sum(int a, int b) {
        int result = simplemath::add(a, b);
        std::ostringstream oss;
        oss << a << " + " << b << " = " << result;
        return oss.str();
    }

    std::string format_product(int a, int b) {
        int result = simplemath::multiply(a, b);
        std::ostringstream oss;
        oss << a << " * " << b << " = " << result;
        return oss.str();
    }

    std::string format_square(int n) {
        int result = simplemath::square(n);
        std::ostringstream oss;
        oss << n << "^2 = " << result;
        return oss.str();
    }

    std::string repeat(const std::string& s, int times) {
        std::string result;
        for (int i = 0; i < times; ++i) {
            result += s;
        }
        return result;
    }
}
```

```cue
    std: "c++20"
}

targets: {
    "multi-deps-example": {
        name: "multi-deps-example"
        type: "executable"
        sources: ["main.cpp"]
        includes: ["vendor"]
        depends: ["stringutils"]
    }
}
```

2. **main.cpp** - Main source (~40-60 lines):
```cpp
#include <iostream>
#include "stringutils/utils.h"
#include "simplemath/math.h"

int main() {
    std::cout << "=== Multi-Deps Example ===" << std::endl;
    std::cout << std::endl;

    // Using stringutils (which uses simplemath internally)
    std::cout << "Using stringutils (chained dep):" << std::endl;
    std::cout << "  " << stringutils::format_sum(10, 5) << std::endl;
    std::cout << "  " << stringutils::format_product(7, 8) << std::endl;
    std::cout << "  " << stringutils::format_square(9) << std::endl;
    std::cout << std::endl;

    // Direct simplemath usage
    std::cout << "Using simplemath directly:" << std::endl;
    std::cout << "  add(100, 200) = " << simplemath::add(100, 200) << std::endl;
    std::cout << std::endl;

    // String utilities
    std::cout << "String repeat: " << stringutils::repeat("abc", 3) << std::endl;

    std::cout << std::endl;
    std::cout << "Multi-deps example complete!" << std::endl;

    return 0;
}
```
