```go
package generate

import (
    "bytes"
    "os"
    "path/filepath"
    "strings"

    "github.com/Duncaen/go-ninja"

// NinjaOptions holds options for generating build.ninja
type NinjaOptions struct {
    Config     *config.Config
    Variants   []string  // Variants to include (e.g., ["debug", "release"])
    BuildDir   string    // e.g., ".build"
    OutputPath string    // Output file path (default: build.ninja)
    Toolchain  string    // "clang" or "gcc"
    Platform   build.Platform
}

// GenerateNinja creates a build.ninja file
func GenerateNinja(opts NinjaOptions) error {
    // ... implementation
}
```

2. Ninja file structure:
   a. Variables section:
      - builddir = .ninja_build (Ninja internal files)
      - cc = clang
      - cxx = clang++
      - ar = ar

   b. Rules section:
      - cc rule: $cc -MD -MF $out.d $cflags -c $in -o $out
        - depfile = $out.d
        - deps = gcc
        - description = CC $out
      - cxx rule: $cxx -MD -MF $out.d $cxxflags -c $in -o $out
        - depfile = $out.d
        - deps = gcc
        - description = CXX $out
      - link rule: $cxx $in -o $out $ldflags
        - description = LINK $out
      - link_shared rule: $cxx -shared $in -o $out $ldflags
        - description = LINK_SHARED $out
      - ar rule: $ar crs $out $in
        - description = AR $out

   c. For each variant:
      - Build statements for each source file with variant-specific flags
      - Link statements for each target
      - Object paths: {builddir}/{variant}/{target}/obj/{source}.o
      - Output paths: {builddir}/{variant}/bin/{target} or lib/lib{target}.a

   d. Phony targets:
      - build debug: phony {all debug outputs}
      - build release: phony {all release outputs}
      - default debug (first variant)

3. Use NinjaPath() for all paths (forward slashes)

4. Stale config detection:
   - Add comment at top: "# Generated by clue. Regenerate with: clue generate ninja"
   - Add build edge that checks clue.cue timestamp (generator = true)
