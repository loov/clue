Create a test project that uses a vendored dependency:

   ```
       └── libmath/
           ├── clue.cue
           ├── math.h
           └── math.cpp
   ```

2. `vendor/libmath/math.cpp`:
   ```cpp
   #include "math.h"

   namespace libmath {
       int add(int a, int b) {
           return a + b;
       }

       int multiply(int a, int b) {
           return a * b;
       }
   }
   ```

3. `vendor/libmath/math.h`:
   ```cpp
   #ifndef LIBMATH_H
   #define LIBMATH_H

   namespace libmath {
       int add(int a, int b);
       int multiply(int a, int b);
   }

   #endif
   ```

4. `vendor/libmath/clue.cue`:
   ```cue
   name: "libmath"
   targets: {
       libmath: {
           name: "libmath"
           type: "static_library"
           sources: ["math.cpp"]
       }
   }
   ```

5. `main.cpp`:
   ```cpp
   #include <iostream>
   #include "libmath/math.h"

   int main() {
       std::cout << "3 + 4 = " << libmath::add(3, 4) << std::endl;
       std::cout << "3 * 4 = " << libmath::multiply(3, 4) << std::endl;
       return 0;
   }
   ```

6. `clue.cue`:
   ```cue
   name: "deps-project"

   dependencies: {
       libmath: {
           type: "vendored"
           path: "vendor/libmath"
       }
   }

   targets: {
       app: {
           name: "app"
           type: "executable"
           sources: ["main.cpp"]
           includes: ["vendor/libmath"]
           depends: ["libmath"]
       }
   }
   ```
  </action>
