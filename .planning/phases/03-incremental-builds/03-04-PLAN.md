```go
// CompilingVerbose reports compilation with reason (verbose mode only)
func (p *Progress) CompilingVerbose(target, filename string, reason RebuildReason, changedFile string) {
    p.current++
    p.built++
    basename := filepath.Base(filename)

    if p.verbose {
        reasonStr := string(reason)
        if changedFile != "" {
            reasonStr = fmt.Sprintf("%s: %s", reason, filepath.Base(changedFile))
        }
        fmt.Fprintf(p.out, "[%d/%d] %s: %s (%s)\n", p.current, p.total, target, basename, reasonStr)
    } else {
        fmt.Fprintf(p.out, "[%d/%d] %s: %s\n", p.current, p.total, target, basename)
    }
}
```

func (p *Progress) Summary() {
    if p.built == 0 && p.cached > 0 {

        return
    }

    if p.cached > 0 {
        fmt.Fprintf(p.out, "Built %d files, %d cached\n", p.built, p.cached)
    } else if p.built > 0 {

    }
}

5. Modify BuildTarget to check cache before compiling each source:
Replace the source compilation loop with cache-aware version:

```go
// In BuildTarget, replace the source compilation loop:

// Get compiler path for cache key
compilerPath := b.compiler.compilerCmd(target.Sources[0])
compilerFullPath, _ := exec.LookPath(compilerPath)
if compilerFullPath == "" {
    compilerFullPath = compilerPath
}

// Compile all source files (with cache check)
var objectFiles []string
for _, source := range target.Sources {
    // Determine object file path
    objName := filepath.Base(source) + ".o"
    objPath := filepath.Join(objDir, objName)

    // Check if rebuild is needed
    needsRebuild, reason, changedFile := b.cacheManager.NeedsRebuild(
        source,
        buildCfg,
        target.Includes,
        compilerFullPath,
        opts.ForceRebuild,
    )

    if !needsRebuild {
        // Skip - use cached result
        progress.Skip(target.Name, source)
        objectFiles = append(objectFiles, objPath)
        continue
    }

    // Report progress with reason (verbose shows why)
    progress.CompilingVerbose(target.Name, source, reason, changedFile)

    // Build compile options
    compileOpts := CompileOptions{
        Source:   source,
        Output:   objPath,
        Includes: target.Includes,
        Defines:  target.Defines,
        Flags:    buildCfg,
        Std:      opts.Config.Toolchain.Std,
    }

    // Compile source
    result, err := b.compiler.CompileSource(ctx, compileOpts)
    if err != nil {
        return &TargetResult{
            Name:     target.Name,
            Type:     target.Type,
            Output:   outputPath,
            Sources:  len(target.Sources),
            Duration: time.Since(start),
            Success:  false,
        }, err
    }

    // Store in cache
    b.cacheManager.StoreResult(
        source,
        objPath,
        result.DepFile,
        buildCfg,
        target.Includes,
        compilerFullPath,
    )

    objectFiles = append(objectFiles, objPath)
}
```
