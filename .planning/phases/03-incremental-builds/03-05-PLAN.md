// createTestProject creates a minimal C++ project for testing
func createTestProject(t *testing.T, dir string) {
    t.Helper()

    // Create source directory
    srcDir := filepath.Join(dir, "src")
    os.MkdirAll(srcDir, 0755)

    // Create header file
    headerContent := `#ifndef CONFIG_H
#define CONFIG_H
#define VERSION 1
#endif
`
    os.WriteFile(filepath.Join(srcDir, "config.h"), []byte(headerContent), 0644)

    // Create main.cpp that includes the header
    mainContent := `#include "config.h"
int getVersion();
int main() { return getVersion() == VERSION ? 0 : 1; }
`
    os.WriteFile(filepath.Join(srcDir, "main.cpp"), []byte(mainContent), 0644)

    // Create utils.cpp that includes the header
    utilsContent := `#include "config.h"
int getVersion() { return VERSION; }
`
    os.WriteFile(filepath.Join(srcDir, "utils.cpp"), []byte(utilsContent), 0644)

    // Create clue.cue config
    configContent := `name: "testproject"
toolchain: {
    compiler: "clang"
    std: "c++17"
}
targets: {
    main: {
        type: "executable"
        sources: ["src/main.cpp", "src/utils.cpp"]
        includes: ["src"]
    }
}
variants: {
    debug: {
        optimization: "none"
        debug_info: true
    }
}
`
    os.WriteFile(filepath.Join(dir, "clue.cue"), []byte(configContent), 0644)
}

// TestIncremental_ContentRevert verifies reverted content reuses cache
func TestIncremental_ContentRevert(t *testing.T) {

    createTestProject(t, tmpDir)

    cfg := loadTestConfig(t, tmpDir)
    builder := NewBuilder("clang", false)

        Verbose:      false,
        Targets:      nil,
        ForceRebuild: false,
    }

    // First build with original content
    _, err := builder.Build(context.Background(), buildOpts)
    if err != nil {
        t.Fatalf("first build failed: %v", err)
    }

    // Save original content
    utilsPath := filepath.Join(tmpDir, "src", "utils.cpp")
    originalContent, _ := os.ReadFile(utilsPath)

    // Modify the file
    os.WriteFile(utilsPath, append(originalContent, []byte("\n// modified\n")...), 0644)

    // Rebuild (will create new cache entry)
    builder2 := NewBuilder("clang", false)
    _, err = builder2.Build(context.Background(), buildOpts)
    if err != nil {
        t.Fatalf("modified build failed: %v", err)
    }

    // Record mtime after modified build
    objDir := filepath.Join(tmpDir, ".build", "debug", "main", "obj")
    utilsObjInfo, _ := os.Stat(filepath.Join(objDir, "utils.cpp.o"))
    modifiedMtime := utilsObjInfo.ModTime()

    // Wait a bit
    time.Sleep(100 * time.Millisecond)

    // Revert to original content
    os.WriteFile(utilsPath, originalContent, 0644)

    // Rebuild - should use original cached result (content hash matches)
    builder3 := NewBuilder("clang", false)
    _, err = builder3.Build(context.Background(), buildOpts)
    if err != nil {
        t.Fatalf("reverted build failed: %v", err)

    // should result in a cache hit if the original was cached.

    // - Reverted build: looks up hash(original), finds it exists

    if err != nil {
        t.Error("utils.cpp.o should exist after revert build")

    if err != nil {
        t.Fatalf("first build failed: %v", err)
    }

    // Record object file mtimes
    objDir := filepath.Join(tmpDir, ".build", "debug", "main", "obj")
    mainObjInfo, _ := os.Stat(filepath.Join(objDir, "main.cpp.o"))
    mainMtime := mainObjInfo.ModTime()

    // Wait a bit
    time.Sleep(100 * time.Millisecond)

    // Add a release variant to config and build with it
    configContent := `name: "testproject"
toolchain: {
    compiler: "clang"
    std: "c++17"
}
targets: {
    main: {
        type: "executable"
        sources: ["src/main.cpp", "src/utils.cpp"]
        includes: ["src"]
    }
}
variants: {
    debug: {
        optimization: "none"
        debug_info: true
    }
    release: {
        optimization: "fast"
        debug_info: false
    }
}
`
    os.WriteFile(filepath.Join(tmpDir, "clue.cue"), []byte(configContent), 0644)

    // Reload config and build with release variant
    cfg2 := loadTestConfig(t, tmpDir)
    cfg2.ActiveVariant = cfg2.Variants["release"]

    buildOpts2 := BuildOptions{
        Config:       cfg2,
        Variant:      "release",
        BuildDir:     filepath.Join(tmpDir, ".build"),
        Verbose:      false,
        Targets:      nil,
        ForceRebuild: false,
