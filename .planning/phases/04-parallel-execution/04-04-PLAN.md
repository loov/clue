// createLargeTestProject creates a 20-file C++ project for parallel testing
func createLargeTestProject(t *testing.T) (projectDir string, cleanup func()) {
    t.Helper()

    dir, err := os.MkdirTemp("", "clue-parallel-test-*")
    require.NoError(t, err)

    // Create 20 source files
    for i := 1; i <= 20; i++ {
        source := fmt.Sprintf(`
#include <iostream>

void func%d() {
    std::cout << "Function %d" << std::endl;
}
`, i, i)
        filename := filepath.Join(dir, fmt.Sprintf("file%02d.cpp", i))
        err := os.WriteFile(filename, []byte(source), 0644)
        require.NoError(t, err)
    }

    // Create main.cpp that calls all functions
    var mainSource strings.Builder
    mainSource.WriteString("#include <iostream>\n\n")
    for i := 1; i <= 20; i++ {
        mainSource.WriteString(fmt.Sprintf("void func%d();\n", i))
    }
    mainSource.WriteString("\nint main() {\n")
    for i := 1; i <= 20; i++ {
        mainSource.WriteString(fmt.Sprintf("    func%d();\n", i))
    }
    mainSource.WriteString("    return 0;\n}\n")
    mainPath := filepath.Join(dir, "main.cpp")
    err = os.WriteFile(mainPath, []byte(mainSource.String()), 0644)
    require.NoError(t, err)

    // Create clue.cue config
    sources := []string{"main.cpp"}
    for i := 1; i <= 20; i++ {
        sources = append(sources, fmt.Sprintf("file%02d.cpp", i))
    }
    cueConfig := fmt.Sprintf(`
name: "parallel-test"
version: "1.0.0"

toolchain: {
    compiler: "clang"
    std: "c++17"
}

targets: {
    "parallel-test": {
        type: "executable"
        sources: %s
    }
}
`, formatCueArray(sources))

    configPath := filepath.Join(dir, "clue.cue")
    err = os.WriteFile(configPath, []byte(cueConfig), 0644)
    require.NoError(t, err)

    cleanup = func() {
        os.RemoveAll(dir)
    }

    return dir, cleanup
}

func formatCueArray(items []string) string {
    var b strings.Builder
    b.WriteString("[")
    for i, item := range items {
        if i > 0 {
            b.WriteString(", ")
        }
        b.WriteString(fmt.Sprintf("%q", item))
    }
    b.WriteString("]")
    return b.String()
}
```

func TestParallelBuild_ScalingComparison(t *testing.T) {
    skipIfNoClang(t)

    projectDir, cleanup := createLargeTestProject(t)
    defer cleanup()

    loader := config.NewLoader()
    cfg, err := loader.Load(projectDir)
    require.NoError(t, err)

    // Clean build with 1 job (sequential)
    builder1 := NewBuilder("clang", false, 1, false)
    opts1 := BuildOptions{
        Config:   cfg,
        Variant:  "debug",
        BuildDir: filepath.Join(projectDir, "build1"),
        Verbose:  false,
        Jobs:     1,
    }

    start1 := time.Now()
    result1, err := builder1.Build(context.Background(), opts1)
    duration1 := time.Since(start1)
    require.NoError(t, err)
    assert.True(t, result1.Success)

    // Clean build with 4 jobs (parallel)
    builder4 := NewBuilder("clang", false, 4, false)
    opts4 := BuildOptions{
        Config:   cfg,
        Variant:  "debug",
        BuildDir: filepath.Join(projectDir, "build4"),
        Verbose:  false,
        Jobs:     4,
    }

    start4 := time.Now()
    result4, err := builder4.Build(context.Background(), opts4)
    duration4 := time.Since(start4)
    require.NoError(t, err)
    assert.True(t, result4.Success)

    // Parallel build should be faster (at least 1.5x speedup with 4 cores)
    // Note: May not be exactly 4x due to linking overhead and I/O
    t.Logf("Sequential (1 job): %v", duration1)
    t.Logf("Parallel (4 jobs): %v", duration4)
    t.Logf("Speedup: %.2fx", float64(duration1)/float64(duration4))

    // Assert parallel is faster (use relaxed threshold for CI variability)
    assert.Less(t, duration4, duration1, "parallel build should be faster than sequential")
}

func TestParallelBuild_KeepGoing(t *testing.T) {
    skipIfNoClang(t)

    dir, err := os.MkdirTemp("", "clue-keepgoing-test-*")
    require.NoError(t, err)
    defer os.RemoveAll(dir)

    // Create project with one file that will fail to compile
    goodSource := `
#include <iostream>
void good_func() {
    std::cout << "Good" << std::endl;
}
`
    badSource := `
this is not valid C++
`
    anotherGood := `
#include <iostream>
void another_good() {
    std::cout << "Another good" << std::endl;
}
`

    os.WriteFile(filepath.Join(dir, "good.cpp"), []byte(goodSource), 0644)
    os.WriteFile(filepath.Join(dir, "bad.cpp"), []byte(badSource), 0644)
    os.WriteFile(filepath.Join(dir, "another.cpp"), []byte(anotherGood), 0644)

    // Create config (as library to avoid link errors)
    cueConfig := `
name: "keepgoing-test"
version: "1.0.0"
toolchain: {
    compiler: "clang"
    std: "c++17"
}
targets: {
    "keepgoing-test": {
        type: "static_library"
        sources: ["good.cpp", "bad.cpp", "another.cpp"]
    }
}
`
    os.WriteFile(filepath.Join(dir, "clue.cue"), []byte(cueConfig), 0644)

    loader := config.NewLoader()
    cfg, err := loader.Load(dir)
    require.NoError(t, err)

    // Build WITHOUT keep-going: should stop on first error
    builder1 := NewBuilder("clang", false, 1, false)
    opts1 := BuildOptions{
        Config:    cfg,
        Variant:   "debug",
        BuildDir:  filepath.Join(dir, "build1"),
        Verbose:   false,
        Jobs:      1,
        KeepGoing: false,
    }

    _, err1 := builder1.Build(context.Background(), opts1)
    assert.Error(t, err1, "build without keep-going should fail")

    // Build WITH keep-going: should compile all valid files
    builder2 := NewBuilder("clang", false, 2, true)
    opts2 := BuildOptions{
        Config:    cfg,
        Variant:   "debug",
        BuildDir:  filepath.Join(dir, "build2"),
        Verbose:   false,
        Jobs:      2,
        KeepGoing: true,
    }

    result, err2 := builder2.Build(context.Background(), opts2)
    // With keep-going, we still get an error but more files compile
    assert.Error(t, err2, "build should still report error")

    // Check that good.cpp and another.cpp compiled successfully
    objDir := filepath.Join(dir, "build2", "debug", "keepgoing-test", "obj")
    _, errGood := os.Stat(filepath.Join(objDir, "good.cpp.o"))
    _, errAnother := os.Stat(filepath.Join(objDir, "another.cpp.o"))

    assert.NoError(t, errGood, "good.cpp should compile even with keep-going")
    assert.NoError(t, errAnother, "another.cpp should compile even with keep-going")

    _ = result // might be nil or partial
}
```
